import { useEffect, useRef, useState } from 'react';
import * as THREE from 'three';
import { NODE_CONFIG } from './nodeConfig';

function geometry(type) {
  if (type === 'decision') return new THREE.OctahedronGeometry(.46);
  if (type === 'flujo-externo') return new THREE.TorusGeometry(.36, .13, 8, 24);
  if (type === 'inicio' || type === 'fin') return new THREE.SphereGeometry(.42, 16, 16);
  return new THREE.BoxGeometry(.78, .46, .46);
}

export default function NeuralMap({ nodes, edges, activeNodeId, onNodeClick }) {
  const container = useRef(null);
  const callbacks = useRef({ activeNodeId, onNodeClick });
  const [tooltip, setTooltip] = useState(null);
  const [failed, setFailed] = useState(false);
  useEffect(() => { callbacks.current = { activeNodeId, onNodeClick }; }, [activeNodeId, onNodeClick]);

  useEffect(() => {
    const element = container.current;
    let renderer;
    try { renderer = new THREE.WebGLRenderer({ antialias: true, alpha: true }); }
    // Initialization failures come from the WebGL renderer, outside React.
    // oxlint-disable-next-line react/set-state-in-effect
    catch { setFailed(true); return; }
    // WebGL is an external system: report renderer initialization to React.
    // oxlint-disable-next-line react/set-state-in-effect
    setFailed(false);
    setTooltip(null);
    renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
    element.appendChild(renderer.domElement);
    const scene = new THREE.Scene();
    const camera = new THREE.PerspectiveCamera(58, 1, .1, 1000);
    scene.add(new THREE.AmbientLight(0xffffff, .7));
    const light = new THREE.PointLight(0xc084fc, 55, 50);
    light.position.set(4, 4, 5); scene.add(light);
    const accent = new THREE.PointLight(0x67e8f9, 30, 40);
    accent.position.set(-4, -3, 3); scene.add(accent);
    const meshes = [], positions = new Map();
    const bounds = new THREE.Box3();
    nodes.forEach(node => {
      const position = new THREE.Vector3(Number(node.posX) || 0, Number(node.posY) || 0, Number(node.posZ) || 0);
      positions.set(node.id, position); bounds.expandByPoint(position);
      const config = NODE_CONFIG[node.tipo] || NODE_CONFIG.paso;
      const mesh = new THREE.Mesh(geometry(node.tipo), new THREE.MeshPhongMaterial({ color: config.color, emissive: config.color, emissiveIntensity: .28, transparent: true, opacity: .9, shininess: 120 }));
      mesh.position.copy(position); mesh.userData.node = node;
      mesh.add(new THREE.Mesh(new THREE.SphereGeometry(.72, 12, 12), new THREE.MeshBasicMaterial({ color: config.color, transparent: true, opacity: .05, side: THREE.BackSide })));
      meshes.push(mesh); scene.add(mesh);
    });
    edges.forEach(edge => {
      const from = positions.get(edge.sourceId), to = positions.get(edge.targetId);
      if (!from || !to) return;
      const middle = from.clone().add(to).multiplyScalar(.5);
      middle.y += edge.condicion === 'negativo' ? -.65 : .65;
      const curve = new THREE.QuadraticBezierCurve3(from, middle, to);
      const color = edge.condicion === 'positivo' ? 0x4ade80 : edge.condicion === 'negativo' ? 0xf87171 : 0xaaaacc;
      const line = new THREE.Line(new THREE.BufferGeometry().setFromPoints(curve.getPoints(24)), edge.tipo === 'modelo' ? new THREE.LineDashedMaterial({ color: 0xbc9aff, dashSize: .15, gapSize: .1 }) : new THREE.LineBasicMaterial({ color, transparent: true, opacity: .6 }));
      line.computeLineDistances(); scene.add(line);
      if (edge.tipo === 'modelo') return;
      const arrow = new THREE.Mesh(new THREE.ConeGeometry(.065, .22, 6), new THREE.MeshBasicMaterial({ color }));
      arrow.position.copy(curve.getPoint(.72));
      arrow.quaternion.setFromUnitVectors(new THREE.Vector3(0, 1, 0), curve.getTangent(.72).normalize());
      scene.add(arrow);
    });
    const center = nodes.length ? bounds.getCenter(new THREE.Vector3()) : new THREE.Vector3();
    const focus = center.clone();
    const size = nodes.length ? bounds.getSize(new THREE.Vector3()).length() : 5;
    const orbit = { theta: .2, phi: Math.PI / 2.4, radius: Math.max(8, size * 1.2) };
    let targetRadius = orbit.radius, frame, lastActive;
    let drag = null, moved = false;
    const ray = new THREE.Raycaster(), mouse = new THREE.Vector2();
    function hit(event) {
      const rect = element.getBoundingClientRect();
      mouse.set((event.clientX - rect.left) / rect.width * 2 - 1, -(event.clientY - rect.top) / rect.height * 2 + 1);
      ray.setFromCamera(mouse, camera);
      return ray.intersectObjects(meshes, false)[0]?.object.userData.node;
    }
    function down(event) {
      drag = { x: event.clientX, y: event.clientY }; moved = false;
      element.setPointerCapture(event.pointerId);
    }
    function move(event) {
      if (drag) {
        const dx = event.clientX - drag.x, dy = event.clientY - drag.y;
        if (Math.abs(dx) + Math.abs(dy) > 2) moved = true;
        orbit.theta -= dx * .005;
        orbit.phi = Math.max(.1, Math.min(Math.PI - .1, orbit.phi + dy * .005));
        drag = { x: event.clientX, y: event.clientY };
        setTooltip(null); return;
      }
      const node = hit(event), rect = element.getBoundingClientRect();
      setTooltip(node ? { node, x: Math.max(4, Math.min(rect.width - 220, event.clientX - rect.left + 14)), y: Math.max(4, Math.min(rect.height - 110, event.clientY - rect.top)) } : null);
      element.style.cursor = node ? 'pointer' : 'grab';
    }
    function up(event) {
      if (drag && !moved) { const node = hit(event); if (node) callbacks.current.onNodeClick?.(node); }
      drag = null;
      if (element.hasPointerCapture(event.pointerId)) element.releasePointerCapture(event.pointerId);
    }
    function cancel() { drag = null; setTooltip(null); }
    function leave() { setTooltip(null); }
    function wheel(event) { event.preventDefault(); targetRadius = Math.max(3, Math.min(Math.max(28, size * 2), targetRadius + event.deltaY * .014)); }
    function resize() {
      const width = Math.max(1, element.clientWidth), height = Math.max(1, element.clientHeight);
      renderer.setSize(width, height); camera.aspect = width / height; camera.updateProjectionMatrix();
    }
    const observer = new ResizeObserver(resize); observer.observe(element); resize();
    const listeners = { pointerdown: down, pointermove: move, pointerup: up, pointercancel: cancel, pointerleave: leave, wheel };
    Object.entries(listeners).forEach(([name, handler]) => element.addEventListener(name, handler, { passive: name !== 'wheel' }));
    function animate(time) {
      frame = requestAnimationFrame(animate);
      const active = callbacks.current.activeNodeId;
      if (active !== lastActive) { if (active) targetRadius = 6; lastActive = active; }
      focus.lerp(positions.get(active) || center, .08);
      orbit.radius += (targetRadius - orbit.radius) * .1;
      camera.position.setFromSphericalCoords(orbit.radius, orbit.phi, orbit.theta).add(focus);
      camera.lookAt(focus);
      meshes.forEach(mesh => {
        const selected = mesh.userData.node.id === active;
        mesh.scale.setScalar(selected ? 1 + Math.sin(time * .003) * .07 : 1);
        mesh.material.emissiveIntensity = selected ? .65 : .28;
        mesh.rotation.y = time * .00018;
      });
      renderer.render(scene, camera);
    }
    frame = requestAnimationFrame(animate);
    return () => {
      cancelAnimationFrame(frame); observer.disconnect();
      Object.entries(listeners).forEach(([name, handler]) => element.removeEventListener(name, handler));
      scene.traverse(object => { object.geometry?.dispose(); object.material?.dispose(); });
      renderer.dispose(); renderer.forceContextLoss(); renderer.domElement.remove();
    };
  }, [nodes, edges]);

  return <div ref={container} style={{ position: 'absolute', inset: 0, touchAction: 'none' }} aria-label="Mapa tridimensional del flujo">
    {failed && <div className="empty" style={{ color: 'white' }}>No se pudo iniciar WebGL. Puedes recorrer el flujo con el botón Recorrer.</div>}
    {tooltip && <div className="glass-heavy" style={{ position: 'absolute', left: tooltip.x, top: tooltip.y, zIndex: 10, padding: 14, borderRadius: 16, width: 210, pointerEvents: 'none' }}>
      <div className="text-accent">{NODE_CONFIG[tooltip.node.tipo]?.label}</div>
      <strong>{tooltip.node.titulo}</strong>
      <p style={{ fontSize: 12, marginTop: 6 }}>{tooltip.node.descripcion}</p>
    </div>}
  </div>;
}

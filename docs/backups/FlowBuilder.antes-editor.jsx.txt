import React, { useState } from 'react';
import { Modal } from './UI';
import { NODE_CONFIG } from './nodeConfig';
import { toast } from '../hooks/useToast';

const NODE_TYPES = Object.entries(NODE_CONFIG).map(([k, v]) => ({ value: k, label: v.emoji + ' ' + v.label }));

function StepRow({ step, idx, total, allFlows, steps, onChange, onDelete }) {
  const isFirst = idx === 0, isLast = idx === total - 1;

  return (
    <div style={{
      display: 'grid', gridTemplateColumns: '36px 1fr auto',
      gap: 10, alignItems: 'flex-start', padding: 14,
      borderRadius: 'var(--r-md)', marginBottom: 10,
      background: 'rgba(255,255,255,0.03)', border: '1px solid rgba(255,255,255,0.06)',
    }}>
      {/* Step number badge */}
      <div style={{
        width: 36, height: 36, borderRadius: 10, display: 'flex', alignItems: 'center',
        justifyContent: 'center', fontWeight: 700, fontSize: 13, color: 'var(--p-300)',
        background: 'rgba(124,58,237,0.18)', border: '1px solid rgba(192,132,252,0.2)',
        flexShrink: 0,
      }}>
        {idx + 1}
      </div>

      {/* Fields */}
      <div style={{ display: 'flex', flexDirection: 'column', gap: 8 }}>
        <div style={{ display: 'flex', gap: 8 }}>
          <select
            className="form-select"
            value={step.tipo}
            disabled={isFirst || isLast}
            onChange={e => onChange({ tipo: e.target.value })}
            style={{ width: 138, fontSize: 12 }}
          >
            {NODE_TYPES.map(t => <option key={t.value} value={t.value}>{t.label}</option>)}
          </select>
          <input
            className="form-input"
            placeholder="Título del paso"
            value={step.titulo}
            onChange={e => onChange({ titulo: e.target.value })}
          />
        </div>

        <input
          className="form-input"
          placeholder="Descripción (opcional)"
          value={step.descripcion}
          onChange={e => onChange({ descripcion: e.target.value })}
          style={{ fontSize: 12 }}
        />

        {/* Decision → condition badges */}
        {step.tipo === 'decision' && (
          <div style={{ display: 'flex', gap: 6, flexWrap: 'wrap', marginTop: 2 }}>
            <span style={{ fontSize: 11, color: 'var(--text-3)', alignSelf: 'center' }}>Salidas:</span>
            {['positivo', 'negativo'].map(c => {
              const on = step.conditions?.includes(c);
              return (
                <button
                  key={c}
                  type="button"
                  onClick={() => {
                    const conds = step.conditions || [];
                    onChange({ conditions: on ? conds.filter(x => x !== c) : [...conds, c] });
                  }}
                  style={{
                    padding: '4px 12px', borderRadius: 20, fontSize: 12, fontWeight: 600,
                    cursor: 'pointer', border: '1px solid', transition: 'all 0.2s',
                    background: on
                      ? c === 'positivo' ? 'rgba(74,222,128,0.15)' : 'rgba(248,113,113,0.15)'
                      : 'rgba(255,255,255,0.04)',
                    color: on
                      ? c === 'positivo' ? 'var(--green)' : 'var(--red)'
                      : 'var(--text-3)',
                    borderColor: on
                      ? c === 'positivo' ? 'rgba(74,222,128,0.3)' : 'rgba(248,113,113,0.3)'
                      : 'rgba(255,255,255,0.07)',
                  }}
                >
                  {c === 'positivo' ? '✓ Positivo' : '✗ Negativo'}
                </button>
              );
            })}
          </div>
        )}

        {/* External flow ref */}
        {step.tipo === 'decision' && step.conditions?.map(condition => (
          <label className="form-label" key={condition}>
            Destino {condition}
            <select aria-label={`Destino ${condition}`} className="form-select" value={step.targets?.[condition] || ''}
              onChange={e => onChange({ targets: { ...step.targets, [condition]: e.target.value } })}>
              <option value="">— Seleccionar destino —</option>
              {steps.filter(s => s._id !== step._id && s.tipo !== 'inicio').map(s => (
                <option key={s._id} value={s._id}>{s.titulo || 'Paso sin título'}</option>
              ))}
            </select>
          </label>
        ))}
        {step.tipo === 'flujo-externo' && (
          <select
            className="form-select"
            value={step.refFlowId || ''}
            onChange={e => onChange({ refFlowId: e.target.value })}
            style={{ fontSize: 12 }}
          >
            <option value="">— Conectar con flujo existente —</option>
            {allFlows.map(f => (
              <option key={f.id} value={f.id}>{f.nombre}</option>
            ))}
          </select>
        )}
      </div>

      {/* Delete (not for first/last) */}
      <button
        type="button"
        className="btn-icon"
        style={{ visibility: (isFirst || isLast) ? 'hidden' : 'visible', alignSelf: 'flex-start', padding: 6 }}
        onClick={onDelete}
        title="Eliminar paso"
      >✕</button>
    </div>
  );
}

export default function FlowBuilder({ processes, allFlows, teamId, onSave, onClose }) {
  const [form, setForm] = useState({ nombre: '', descripcion: '', processId: '', version: '1.0' });
  const [steps, setSteps] = useState([
    { _id: 1, tipo: 'inicio',  titulo: 'Inicio',  descripcion: '', conditions: [] },
    { _id: 2, tipo: 'paso',    titulo: '',         descripcion: '', conditions: [] },
    { _id: 3, tipo: 'fin',     titulo: 'Fin',      descripcion: '', conditions: [] },
  ]);
  const [saving, setSaving] = useState(false);

  const teamProcesses = processes.filter(p => p.teamId === teamId);

  const addStep = () => {
    setSteps(prev => {
      const last = prev[prev.length - 1];
      const body = prev.slice(0, -1);
      return [...body, { _id: crypto.randomUUID(), tipo: 'paso', titulo: '', descripcion: '', conditions: [] }, last];
    });
  };

  const updateStep = (id, patch) =>
    setSteps(prev => prev.map(s => s._id === id ? { ...s, ...patch } : s));

  const deleteStep = (id) =>
    setSteps(prev => prev.filter(s => s._id !== id));

  const handleSave = async () => {
    if (!form.nombre.trim())    { toast.error('El flujo necesita un nombre');       return; }
    if (!form.processId)        { toast.error('Selecciona el proceso al que pertenece'); return; }
    if (steps.some(s => !s.titulo.trim() && s.tipo !== 'inicio')) {
      toast.error('Todos los pasos deben tener un título'); return;
    }
    if (steps.some(s => s.tipo === 'decision' && (!s.conditions?.length || s.conditions.some(c => !steps.some(target => String(target._id) === String(s.targets?.[c]) && target._id !== s._id && target.tipo !== 'inicio'))))) {
      toast.error('Selecciona las salidas y sus destinos para cada decisión'); return;
    }
    if (steps.some(s => s.tipo === 'flujo-externo' && !s.refFlowId)) {
      toast.error('Selecciona el flujo de destino de cada paso externo'); return;
    }
    setSaving(true);
    try {
      // Convert steps → nodes with 3D positions
      const nodes = steps.map((s, i) => {
        const col = i % 3, row = Math.floor(i / 3);
        return {
          _tempId:     s._id,
          tipo:        s.tipo,
          titulo:      s.titulo || `Paso ${i + 1}`,
          descripcion: s.descripcion || '',
          posX:        (col - 1) * 2.8,
          posY:        (1 - row) * 2.2,
          posZ:        (Math.random() - 0.5) * 0.6,
          refFlowId:   s.refFlowId || '',
          metadata:    { conditions: s.conditions || [] },
        };
      });

      // Build sequential edges (with condition slots for decision nodes)
      const edges = [];
      for (let i = 0; i < steps.length - 1; i++) {
        const cur  = steps[i];
        const next = steps[i + 1];
        if (cur.tipo === 'decision' && cur.conditions?.length > 0) {
          cur.conditions.forEach(c => {
            edges.push({ sourceId: cur._id, targetId: cur.targets[c], condicion: c, etiqueta: c === 'positivo' ? 'Sí' : 'No' });
          });
        } else {
          if (cur.tipo !== 'fin') edges.push({ sourceId: cur._id, targetId: next._id, condicion: 'siempre', etiqueta: '' });
        }
      }

      await onSave({ flow: { ...form, teamId }, nodes, edges });
    } catch (err) {
      toast.error(err.message);
    } finally {
      setSaving(false);
    }
  };

  return (
    <Modal title="➕ Crear Nuevo Flujo" onClose={onClose} wide>
      {/* Flow metadata */}
      <div style={{ display: 'flex', flexDirection: 'column', gap: 14 }}>
        <div className="form-row">
          <div className="form-group">
            <label className="form-label">Nombre del Flujo</label>
            <input
              className="form-input"
              placeholder="Ej: Atención de Ticket Técnico"
              value={form.nombre}
              onChange={e => setForm(p => ({ ...p, nombre: e.target.value }))}
            />
          </div>
          <div className="form-group">
            <label className="form-label">Proceso</label>
            <select
              className="form-select"
              value={form.processId}
              onChange={e => setForm(p => ({ ...p, processId: e.target.value }))}
            >
              <option value="">— Seleccionar —</option>
              {teamProcesses.map(p => <option key={p.id} value={p.id}>{p.nombre}</option>)}
            </select>
          </div>
        </div>

        <div className="form-row">
          <div className="form-group">
            <label className="form-label">Descripción</label>
            <input
              className="form-input"
              placeholder="¿Qué resuelve este flujo?"
              value={form.descripcion}
              onChange={e => setForm(p => ({ ...p, descripcion: e.target.value }))}
            />
          </div>
          <div className="form-group">
            <label className="form-label">Versión</label>
            <input
              className="form-input"
              placeholder="1.0"
              value={form.version}
              onChange={e => setForm(p => ({ ...p, version: e.target.value }))}
            />
          </div>
        </div>

        {/* Steps */}
        <div style={{ fontSize: 10, fontWeight: 700, textTransform: 'uppercase', letterSpacing: '1px', color: 'var(--text-3)', marginTop: 4 }}>
          📋 Pasos del Flujo
        </div>
        <div style={{ maxHeight: '38vh', overflowY: 'auto', paddingRight: 4 }}>
          {steps.map((step, idx) => (
            <StepRow
              key={step._id}
              step={step}
              idx={idx}
              total={steps.length}
              allFlows={allFlows}
              steps={steps}
              onChange={patch => updateStep(step._id, patch)}
              onDelete={() => deleteStep(step._id)}
            />
          ))}
        </div>
        <button
          type="button"
          className="btn btn-secondary btn-sm"
          onClick={addStep}
          style={{ alignSelf: 'flex-start' }}
        >
          + Agregar paso
        </button>
      </div>

      <div className="modal-footer">
        <button className="btn btn-secondary" onClick={onClose}>Cancelar</button>
        <button className="btn btn-primary" onClick={handleSave} disabled={saving}>
          {saving ? '💾 Guardando...' : '💾 Crear Flujo'}
        </button>
      </div>
    </Modal>
  );
}

// ID de tu hoja de cálculo (Base de datos)
const SPREADSHEET_ID = '12-RXAOuu_sVS479CdrhjBeKO9Tdej-93lhe6gPEnku0';

// Nombres de las hojas (pestañas)
const SHEET_NAMES = {
  TEAMS: 'Equipos',
  PROCESSES: 'Procesos',
  FLOWS: 'Flujos',
  NODES: 'Nodos',
  EDGES: 'Conexiones'
};

// =========================================================
// CONFIGURACIÓN INICIAL
// =========================================================

/**
 * Función para configurar la base de datos inicial.
 * Crea las hojas y los encabezados si no existen.
 */
function setupSpreadsheet() {
  // Ahora abrimos el archivo específicamente con tu ID
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);

  const schemas = {
    [SHEET_NAMES.TEAMS]:     ['id','nombre','descripcion','color','icono','creadoEn','actualizadoEn'],
    [SHEET_NAMES.PROCESSES]: ['id','teamId','nombre','descripcion','orden','creadoEn','actualizadoEn'],
    [SHEET_NAMES.FLOWS]:     ['id','processId','teamId','nombre','descripcion','version','estado','creadoEn','actualizadoEn'],
    [SHEET_NAMES.NODES]:     ['id','flowId','tipo','titulo','descripcion','posX','posY','posZ','metadata','creadoEn'],
    [SHEET_NAMES.EDGES]:     ['id','flowId','sourceId','targetId','condicion','etiqueta','creadoEn']
  };

  const created = [];
  Object.entries(schemas).forEach(([name, headers]) => {
    let sheet = ss.getSheetByName(name);
    if (!sheet) {
      sheet = ss.insertSheet(name);
      created.push(name);
    }
    // Encabezados si la hoja está vacía
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(headers);
      // Formato encabezado
      const headerRange = sheet.getRange(1, 1, 1, headers.length);
      headerRange.setBackground('#1a1a2e')
                 .setFontColor('#c084fc')
                 .setFontWeight('bold')
                 .setFontSize(11);
      sheet.setFrozenRows(1);
    }
    // Auto-resize columnas
    sheet.autoResizeColumns(1, headers.length);
  });

  return { message: 'Setup completado', hojas: created };
}

// =========================================================
// ENDPOINTS DE LA API (GET Y POST)
// =========================================================

/**
 * Método GET: Se usará para LEER información desde React (Ej. Cargar el mapa 3D).
 */
function doGet(e) {
  try {
    const action = e.parameter.action;
    const sheetName = e.parameter.sheet;
    
    if (action === 'getData' && sheetName) {
      const data = getSheetDataAsJSON(sheetName);
      return createJsonResponse({ success: true, data: data });
    }
    
    return createJsonResponse({ success: false, message: 'Acción GET no válida' });
  } catch (error) {
    return createJsonResponse({ success: false, error: error.toString() });
  }
}

/**
 * Método POST: Se usará para GUARDAR información desde React (Ej. Crear flujos, equipos).
 */
function doPost(e) {
  try {
    const requestData = JSON.parse(e.postData.contents);
    const action = requestData.action;
    const sheetName = requestData.sheet;
    const payload = requestData.payload; 

    if (action === 'insertRow' && sheetName && payload) {
      insertDataToSheet(sheetName, payload);
      return createJsonResponse({ success: true, message: 'Registro creado con éxito' });
    }

    return createJsonResponse({ success: false, message: 'Acción POST no válida' });
  } catch (error) {
    return createJsonResponse({ success: false, error: error.toString() });
  }
}

// FUNCIONES AUXILIARES (HELPERS)
// =========================================================

/**
 * Lee todos los datos de una hoja usando el ID del archivo y los convierte en JSON.
 */
function getSheetDataAsJSON(sheetName) {
  // Abrimos usando el ID
  const sheet = SpreadsheetApp.openById(SPREADSHEET_ID).getSheetByName(sheetName);
  if (!sheet) return [];
  
  const lastRow = sheet.getLastRow();
  const lastCol = sheet.getLastColumn();
  
  if (lastRow < 2) return []; // Solo hay encabezados
  
  const dataRange = sheet.getRange(1, 1, lastRow, lastCol).getValues();
  const headers = dataRange[0];
  const rows = dataRange.slice(1);
  
  const jsonData = rows.map(row => {
    let obj = {};
    headers.forEach((header, index) => {
      obj[header] = row[index];
    });
    return obj;
  });
  
  return jsonData;
}

/**
 * Inserta una nueva fila de datos en la hoja especificada usando el ID del archivo.
 */
function insertDataToSheet(sheetName, payload) {
  // Abrimos usando el ID
  const sheet = SpreadsheetApp.openById(SPREADSHEET_ID).getSheetByName(sheetName);
  if (!sheet) throw new Error("La hoja no existe: " + sheetName);

  const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
  const newRow = [];

  headers.forEach(header => {
    newRow.push(payload[header] !== undefined ? payload[header] : "");
  });

  sheet.appendRow(newRow);
}

/**
 * Crea una respuesta estándar en formato JSON requerida por React.
 */
function createJsonResponse(responseObject) {
  return ContentService.createTextOutput(JSON.stringify(responseObject))
    .setMimeType(ContentService.MimeType.JSON);
}
